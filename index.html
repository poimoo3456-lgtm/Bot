Flow การสั่งซื้อ elif text == "🛒 สั่งซื้อ": bot.send_message(chat_id, "เลือกบริการที่สนใจ:", reply_markup=get_order_menu()) elif text in ["🤖 AI Chatbot", "🌐 เว็บไซต์", "📱 แอปมือถือ", "🏢 ระบบบริษัท"]: user_state[chat_id] = {"step": "NAME", "product": text} bot.send_message(chat_id, f"คุณเลือก {text}\nรบกวนขอทราบ 'ชื่อ' ของคุณครับ:") #
